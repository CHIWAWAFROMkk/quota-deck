# QuotaDeck 贴边 HUD（第一期）Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 QuotaDeck 里加一条贴屏幕边缘的深色 HUD：每个服务商一个额度圆环，悬停出一行明细，点击出卡片（额度、消耗速率、24 小时 Token 柱），可拖到任意屏幕边；同时让 Claude 额度改读官方用量接口。

**Architecture:** 主进程新增四个纯模块（`claude-oauth`、`token-ledger`、`hud-geometry`、`hud-view`）和一个窗口模块（`hud-window`）。HUD 是单个固定 480×480、透明、置顶、默认鼠标穿透的 `BrowserWindow`，浮条和卡片都在里面用 CSS 画；主进程每 60 秒把 `readAll()` 的 UI 快照转成 HUD 条目推过去，Token 统计单独推送。渲染进程只展示，不读文件、不联网。

**Tech Stack:** Electron 44.4.5（CommonJS `.cjs`）、Node 内置 `node:test`、原生 DOM/SVG，无新增依赖。

**Spec:** `docs/superpowers/specs/2026-10-13-edge-hud-design.md`

```js
const fs = require('node:fs/promises');
const os = require('node:os');
const path = require('node:path');

const USAGE_URL = 'https://api.anthropic.com/api/oauth/usage';
const OAUTH_SOURCE = 'Claude 官方用量接口';
const LABELS = { five_hour: '5 小时', seven_day: '每周', seven_day_opus: '每周 · Opus', seven_day_sonnet: '每周 · Sonnet' };
const ORDER = Object.keys(LABELS);
const REASONS = {
  'no-credentials': '没找到 Claude 登录信息，先在 Claude Code 里登录。',
  expired: '登录令牌过期，在 Claude Code 里发一条消息即可续期。',
  'rate-limited': 'Claude 用量接口限流，稍后自动重试。',
  network: '连不上 Claude 用量接口，检查网络或代理。',
  'bad-response': 'Claude 用量接口返回了无法识别的数据。',
};
const BACKOFF_START = 2 * 60_000;
const BACKOFF_MAX = 15 * 60_000;

function defaultCredentialsPath() {
  return path.join(os.homedir(), '.claude', '.credentials.json');
}

// Read-only. Claude Code renews the token and rewrites this file, so read it on every refresh.
async function readCredentials(file) {
  let text;
  try { text = await fs.readFile(file, 'utf8'); } catch { return null; }
  try {
    const o = JSON.parse(text.replace(/^\uFEFF/, ''))?.claudeAiOauth;
    if (typeof o?.accessToken !== 'string' || !o.accessToken) return null;
    return { token: o.accessToken, plan: typeof o.subscriptionType === 'string' ? o.subscriptionType : '' };
  } catch {
    return null; // Claude Code may be rewriting it; the next refresh reads it again.
  }
}

const rank = id => (ORDER.includes(id) ? ORDER.indexOf(id) : ORDER.length);
const labelFor = id => LABELS[id] || `每周 · ${id.replace(/^seven_day_/, '')}`;

// Only five_hour / seven_day* are quota windows; extra_usage and other fields are not.
function parseUsage(json, plan, now) {
  if (!json || typeof json !== 'object' || Array.isArray(json)) return null;
  const windows = [];
  for (const [id, value] of Object.entries(json)) {
    if (!/^(five_hour|seven_day)/.test(id)) continue;
    if (!value || typeof value !== 'object' || typeof value.utilization !== 'number' || !Number.isFinite(value.utilization)) continue;
    const at = typeof value.resets_at === 'string' ? Date.parse(value.resets_at) : NaN;
    const used = Math.max(0, Math.min(100, value.utilization));
    windows.push({ id, label: labelFor(id), remainingPercent: Math.round((100 - used) * 1000) / 1000, resetsAt: Number.isFinite(at) ? new Date(at).toISOString() : null });
  }
  if (!windows.some(w => w.id === 'five_hour')) return null;
  windows.sort((a, b) => rank(a.id) - rank(b.id));
  const remainingPercent = Math.min(...windows.map(w => w.remainingPercent));
  return {
    id: 'claude',
    name: 'Claude Code',
    precision: 'provider-reported',
    status: remainingPercent <= 10 ? 'critical' : remainingPercent <= 30 ? 'low' : 'healthy',
    updatedAt: new Date(now).toISOString(),
    source: OAUTH_SOURCE,
    planType: plan,
    remainingPercent,
    windows,
    models: [],
    groups: [{
      displayName: 'Claude 订阅共享额度',
      buckets: windows.map(w => ({
        bucketId: w.id,
        label: w.label,
        window: w.id === 'five_hour' ? '5h' : 'weekly',
        remainingFraction: w.remainingPercent / 100,
        resetTime: w.resetsAt,
      })),
    }],
    quotaNote: '订阅窗口 · 官方用量接口',
    note: '由 Claude 官方用量接口读取，与 Claude Code 的 /usage 同源。',
  };
}

function defaultFetch() {
  // In the Electron main process net.fetch follows the system proxy; plain Node falls back to fetch.
  const electron = require('electron');
  return typeof electron === 'object' && electron?.net?.fetch ? electron.net.fetch.bind(electron.net) : globalThis.fetch;
}

function createClaudeOAuthReader({ credentialsPath = defaultCredentialsPath(), fetchImpl = null, now = Date.now, timeoutMs = 15_000 } = {}) {
  let backoff = 0;
  let blockedUntil = 0;
  return async function readClaudeOAuth() {
    const t = now();
    if (t < blockedUntil) return { ok: false, reason: 'rate-limited' };
    const credentials = await readCredentials(credentialsPath);
    if (!credentials) return { ok: false, reason: 'no-credentials' };
    let response;
    try {
      response = await (fetchImpl || defaultFetch())(USAGE_URL, {
        method: 'GET',
        headers: { Authorization: `Bearer ${credentials.token}`, 'anthropic-beta': 'oauth-2025-04-20', Accept: 'application/json' },
        signal: AbortSignal.timeout(timeoutMs),
      });
    } catch {
      return { ok: false, reason: 'network' }; // never forward the error: it may echo request details
    }
    if (response.status === 401 || response.status === 403) return { ok: false, reason: 'expired' };
    if (response.status === 429) {
      backoff = Math.min(BACKOFF_MAX, backoff ? backoff * 2 : BACKOFF_START);
      blockedUntil = t + backoff;
      return { ok: false, reason: 'rate-limited' };
    }
    if (!response.ok) return { ok: false, reason: 'network' };
    let json;
    try { json = await response.json(); } catch { return { ok: false, reason: 'bad-response' }; }
    const snapshot = parseUsage(json, credentials.plan, t);
    if (!snapshot) return { ok: false, reason: 'bad-response' };
    backoff = 0;
    blockedUntil = 0;
    return { ok: true, snapshot };
  };
}

module.exports = { OAUTH_SOURCE, REASONS, USAGE_URL, createClaudeOAuthReader, parseUsage, readCredentials };
```

```js
const fs = require('node:fs/promises');
const os = require('node:os');
const path = require('node:path');

const QUARTER = 15 * 60000;
const BUCKETS = 96;
const CHUNK = 4 * 1024 * 1024;
const MAX_LINE = 1024 * 1024;
const MAX_DEPTH = 5;

/** Start of the oldest of 96 local quarter-hour buckets; the newest one contains `now`. */
function windowStart(now) {
  const d = new Date(now);
  d.setSeconds(0, 0);
  d.setMinutes(d.getMinutes() - (d.getMinutes() % 15));
  return d.getTime() - (BUCKETS - 1) * QUARTER;
}

const count = (usage, key) => (Number.isFinite(usage[key]) && usage[key] > 0 ? usage[key] : 0);

// One assistant message can span several lines; the caller keeps the last one per id.
function parseClaudeLine(line) {
  if (!line.includes('"usage"')) return null;
  let row;
  try { row = JSON.parse(line); } catch { return null; }
  const usage = row?.message?.usage;
  const id = row?.message?.id;
  const at = Date.parse(row?.timestamp);
  if (row?.type !== 'assistant' || typeof id !== 'string' || !usage || typeof usage !== 'object' || !Number.isFinite(at)) return null;
  return { id, at, tokens: count(usage, 'input_tokens') + count(usage, 'output_tokens') + count(usage, 'cache_creation_input_tokens') + count(usage, 'cache_read_input_tokens') };
}

function parseCodexLine(line) {
  if (!line.includes('"token_count"')) return null;
  let row;
  try { row = JSON.parse(line); } catch { return null; }
  const total = row?.payload?.info?.total_token_usage?.total_tokens;
  const at = Date.parse(row?.timestamp);
  if (row?.type !== 'event_msg' || row?.payload?.type !== 'token_count' || !Number.isFinite(total) || total < 0 || !Number.isFinite(at)) return null;
  return { at, total };
}

async function listRecentJsonl(root, since, fileSystem) {
  const found = [];
  async function walk(dir, depth) {
    let entries;
    try { entries = await fileSystem.readdir(dir, { withFileTypes: true }); }
    catch (error) { if (depth === 0) throw error; return; }
    for (const entry of entries) {
      const full = path.join(dir, entry.name);
      if (entry.isDirectory()) { if (depth < MAX_DEPTH) await walk(full, depth + 1); continue; }
      if (!entry.isFile() || !entry.name.endsWith('.jsonl')) continue;
      try {
        const stat = await fileSystem.stat(full);
        if (stat.mtimeMs >= since) found.push({ file: full, stat });
      } catch { /* deleted between readdir and stat */ }
    }
  }
  await walk(root, 0);
  return found;
}

// Reads complete lines after state.offset; a trailing partial line waits for the next read.
async function readLines(file, state, size, fileSystem, onLine) {
  if (size <= state.offset) return;
  const handle = await fileSystem.open(file, 'r');
  try {
    let position = state.offset;
    let carry = Buffer.alloc(0);
    while (position < size) {
      const chunk = Buffer.alloc(Math.min(CHUNK, size - position));
      const { bytesRead } = await handle.read(chunk, 0, chunk.length, position);
      if (!bytesRead) break;
      position += bytesRead;
      const data = carry.length ? Buffer.concat([carry, chunk.subarray(0, bytesRead)]) : chunk.subarray(0, bytesRead);
      let start = 0;
      for (let end = data.indexOf(0x0a); end >= 0; end = data.indexOf(0x0a, start)) {
        if (end - start <= MAX_LINE) onLine(data.toString('utf8', start, end));
        start = end + 1;
      }
      carry = Buffer.from(data.subarray(start));
    }
    state.offset = position - carry.length;
  } finally {
    await handle.close();
  }
}

const SOURCES = {
  claude: {
    fresh: () => ({ offset: 0, mtimeMs: 0, events: new Map() }),
    take(state, line) {
      const event = parseClaudeLine(line);
      if (event) state.events.set(event.id, event);
    },
    events: state => state.events.values(),
    prune(state, since) { for (const [id, event] of state.events) if (event.at < since) state.events.delete(id); },
  },
  codex: {
    fresh: () => ({ offset: 0, mtimeMs: 0, lastTotal: 0, events: [] }),
    take(state, line) {
      const event = parseCodexLine(line);
      if (!event) return;
      // A cumulative total going down starts a new segment counted from zero.
      const delta = event.total >= state.lastTotal ? event.total - state.lastTotal : event.total;
      state.lastTotal = event.total;
      if (delta > 0) state.events.push({ at: event.at, tokens: delta });
    },
    events: state => state.events,
    prune(state, since) { state.events = state.events.filter(event => event.at >= since); },
  },
};

function emptySeries(status, startAt) {

    const source = SOURCES[id];
    const known = files[id];
    const series = emptySeries('ok', startAt);
    let recent;
    try { recent = await listRecentJsonl(roots[id], startAt, fileSystem); }
    catch (error) { return emptySeries(error?.code === 'ENOENT' ? 'unsupported' : 'error', startAt); }
    const alive = new Set();
    for (const { file, stat } of recent) {
      alive.add(file);
      if (clock() > deadline) { series.partial = true; continue; } // keep what earlier rounds parsed
      let state = known.get(file);
      if (!state || stat.size < state.offset || stat.mtimeMs < state.mtimeMs) {
        state = source.fresh();
        known.set(file, state);
      }
      try {
        await readLines(file, state, stat.size, fileSystem, line => source.take(state, line));
        state.mtimeMs = stat.mtimeMs;
        source.prune(state, startAt);
      } catch {
        known.delete(file); // reread from scratch next round rather than double count
        series.partial = true;
      }
    }
    for (const file of [...known.keys()]) if (!alive.has(file)) known.delete(file);
    const seen = new Set();
    for (const state of known.values()) {
      for (const event of source.events(state)) {
        if (event.id) { if (seen.has(event.id)) continue; seen.add(event.id); }
        const index = Math.floor((event.at - startAt) / QUARTER);
        if (index < 0 || index >= BUCKETS) continue;
        series.buckets[index] += event.tokens;
        series.total += event.tokens;
      }
    }
    return series;
  }

  async function read(now = clock()) {
    const startAt = windowStart(now);
    const deadline = clock() + budgetMs;
    return { claude: await readSource('claude', startAt, deadline), codex: await readSource('codex', startAt, deadline) };
  }

  return {
    read(now) {
      if (!pending) pending = read(now).finally(() => { pending = null; });
      return pending;
    },
  };
}

module.exports = { BUCKETS, QUARTER, createTokenLedger, parseClaudeLine, parseCodexLine, windowStart };
```

```js
const MARGIN = 3;
const WINDOW_SIZE = 480;
// Five 22px rings, 8px gaps, 8px end padding and 6px side padding. Must match hud.css.
const STRIP = { long: 158, short: 34 };
const EDGES = ['left', 'right', 'top', 'bottom'];
const DEFAULT_SETTINGS = Object.freeze({ displayId: null, edge: 'right', offset: 0.3, visible: true });

const clamp = (value, low, high) => Math.min(high, Math.max(low, value));
const isVertical = edge => edge === 'left' || edge === 'right';

function normalizeHudSettings(raw) {
  const r = raw && typeof raw === 'object' && !Array.isArray(raw) ? raw : {};
  return {
    displayId: Number.isSafeInteger(r.displayId) ? r.displayId : null,
    edge: EDGES.includes(r.edge) ? r.edge : DEFAULT_SETTINGS.edge,
    offset: Number.isFinite(r.offset) ? clamp(r.offset, 0, 1) : DEFAULT_SETTINGS.offset,
    visible: typeof r.visible === 'boolean' ? r.visible : DEFAULT_SETTINGS.visible,
  };
}

/** Window rectangle in screen coordinates, and the collapsed strip's top-left inside it. */
function layoutHud({ workArea: wa, edge, offset }) {
  const width = Math.min(WINDOW_SIZE, wa.width);
  const height = Math.min(WINDOW_SIZE, wa.height);
  const along = clamp(Number.isFinite(offset) ? offset : DEFAULT_SETTINGS.offset, 0, 1);
  let sx; let sy; let wx; let wy;
  if (isVertical(edge)) {
    sy = Math.round(wa.y + along * Math.max(0, wa.height - STRIP.long));
    sx = edge === 'left' ? wa.x + MARGIN : wa.x + wa.width - MARGIN - STRIP.short;
    wx = edge === 'left' ? wa.x : wa.x + wa.width - width;
    wy = Math.round(clamp(sy + STRIP.long / 2 - height / 2, wa.y, wa.y + wa.height - height));
  } else {
    sx = Math.round(wa.x + along * Math.max(0, wa.width - STRIP.long));
    sy = edge === 'top' ? wa.y + MARGIN : wa.y + wa.height - MARGIN - STRIP.short;
    wy = edge === 'top' ? wa.y : wa.y + wa.height - height;
    wx = Math.round(clamp(sx + STRIP.long / 2 - width / 2, wa.x, wa.x + wa.width - width));
  }
  return { window: { x: wx, y: wy, width, height }, strip: { x: sx - wx, y: sy - wy }, edge };
}

/** Nearest edge to the strip's centre, and the strip's position along that edge (0–1). */
function snapToEdge({ workArea: wa, point }) {
  const px = clamp(point.x, wa.x, wa.x + wa.width);
  const py = clamp(point.y, wa.y, wa.y + wa.height);
  const distance = { left: px - wa.x, right: wa.x + wa.width - px, top: py - wa.y, bottom: wa.y + wa.height - py };
  const edge = EDGES.reduce((best, candidate) => (distance[candidate] < distance[best] ? candidate : best), EDGES[0]);
  const along = isVertical(edge)
    ? (py - wa.y - STRIP.long / 2) / Math.max(1, wa.height - STRIP.long)
    : (px - wa.x - STRIP.long / 2) / Math.max(1, wa.width - STRIP.long);
  return { edge, offset: clamp(along, 0, 1) };
}

/** The saved display if it is still connected, otherwise the primary one. */
function pickDisplay(displays, primary, displayId) {
  return displays.find(display => display.id === displayId) || primary;
}

module.exports = { DEFAULT_SETTINGS, EDGES, MARGIN, STRIP, WINDOW_SIZE, layoutHud, normalizeHudSettings, pickDisplay, snapToEdge };
```

```js
const HUD_ORDER = ['codex', 'claude', 'antigravity', 'deepseek', 'workbuddy'];
const NAMES = { codex: 'Codex', claude: 'Claude Code', antigravity: 'Antigravity', deepseek: 'DeepSeek', workbuddy: 'WorkBuddy' };
const TOKEN_SOURCES = ['claude', 'codex'];
const WEEKDAYS = ['周日', '周一', '周二', '周三', '周四', '周五', '周六'];
const pad = n => String(n).padStart(2, '0');

function levelFor(usedPct) {
  if (!Number.isFinite(usedPct)) return 'none';
  return usedPct >= 90 ? 'high' : usedPct >= 70 ? 'warn' : 'ok';
}

const usedOf = remaining => (Number.isFinite(remaining) ? Math.round(100 - Math.max(0, Math.min(100, remaining))) : null);

function shortLabel(label) {
  const text = String(label || '');
  if (text === '5 小时') return '5h';
  if (text === '每周') return '周';
  const named = /^每周 · (.+)$/.exec(text);
  return named ? named[1] : text;
}

function resetText(iso, now) {
  const at = typeof iso === 'string' ? Date.parse(iso) : NaN;
  if (!Number.isFinite(at)) return '重置时间未知';
  if (at <= now) return '待重置';
  const d = new Date(at);
  const hm = `${pad(d.getHours())}:${pad(d.getMinutes())}`;
  return at - now < 86_400_000 ? `${hm} 重置` : `${WEEKDAYS[d.getDay()]} ${hm} 重置`;
}

const burnText = burn => (Number.isFinite(burn) ? `${Math.max(0, burn).toFixed(1)}%/h` : '采样中');

// Pools carry per-pool windows (Claude, Antigravity); otherwise use the gauge's windows (Codex).
function windowRows(provider, now) {
  const pools = (provider.pools || []).filter(pool => pool.windows?.length);
  const windows = pools.length
    ? pools.flatMap(pool => pool.windows.map(w => ({ ...w, pool: pools.length > 1 ? pool.name : '' })))
    : (provider.gauge?.windows || []).map(w => ({ ...w, pool: '' }));
  return windows.map(w => {
    const usedPct = usedOf(w.remainingPercent);
    return {
      label: w.pool ? `${w.pool} · ${w.label}` : w.label,
      short: w.pool ? `${w.pool} ${shortLabel(w.label)}` : shortLabel(w.label),
      usedPct,
      level: levelFor(usedPct),
      reset: resetText(w.resetsAt, now),
      burn: burnText(w.burnPerHour),
    };
  });
}

function placeholder(id) {
  return {
    id, name: NAMES[id], logo: `${id}.svg`,
    ring: { usedPct: null, level: 'none', dashed: false },
    label: '–',
    pill: [{ name: '', text: '未读取', level: 'none' }],
    card: { plan: '', freshness: '', note: '还没有读取到这个服务商的数据。', stale: false, balance: null, rows: [], tokens: TOKEN_SOURCES.includes(id) },
  };
}

function buildHudItems(ui, now = Date.now()) {
  const providers = new Map((ui?.providers || []).map(provider => [provider.id, provider]));
  return HUD_ORDER.map(id => {
    const p = providers.get(id);
    if (!p) return placeholder(id);
    const gauge = p.gauge || { kind: 'unknown', provenance: 'unknown' };
    const stale = gauge.provenance === 'stale';
    const rows = windowRows(p, now);
    let ring = { usedPct: null, level: 'none', dashed: false };
    let label = '–';
    let balance = null;
    if (gauge.kind === 'window' || gauge.kind === 'ratio') {
      const usedPct = usedOf(gauge.remainingPercent);
      ring = { usedPct, level: levelFor(usedPct), dashed: stale };
      if (usedPct !== null) label = `${usedPct}%`;
    } else if (gauge.kind === 'balance') {
      balance = String(p.quotaHero || '').replace(/\s+/g, '') || null;
      ring = { usedPct: null, level: p.status === 'critical' ? 'high' : 'ok', dashed: stale };
      label = balance || '–';
    }
    const pill = rows.length
      ? rows.map(row => ({ name: row.short, text: row.usedPct === null ? '–' : `${row.usedPct}%`, level: row.level }))
      : [{ name: '', text: balance ? `余额 ${balance}` : gauge.kind === 'ratio' ? `已用 ${label}` : (p.statusText || '未知'), level: ring.level }];
    const note = gauge.kind === 'unknown' ? (p.quotaSub || p.statusText || '') : stale ? '旧数据，不代表当前额度。' : '';
    return {
      id, name: p.name || NAMES[id], logo: `${id}.svg`, ring, label, pill,
      card: { plan: p.plan || '', freshness: p.freshness || '', note, stale, balance, rows, tokens: TOKEN_SOURCES.includes(id) },
    };
  });
}

module.exports = { HUD_ORDER, buildHudItems, levelFor, resetText };
```

```js
const { contextBridge, ipcRenderer } = require('electron');

function subscribe(channel, callback) {
  const listener = (_event, value) => callback(value);
  ipcRenderer.on(channel, listener);
  return () => ipcRenderer.removeListener(channel, listener);
}

contextBridge.exposeInMainWorld('hud', {

  refresh: () => ipcRenderer.invoke('hud:refresh'),
```

- [ ] **Step 4: 写 `hud-window.cjs`**

```js
const fs = require('node:fs');
const path = require('node:path');
const { randomUUID } = require('node:crypto');
const { BrowserWindow, ipcMain, screen } = require('electron');
const { isTrustedRenderer } = require('./security.cjs');
const { STRIP, WINDOW_SIZE, layoutHud, normalizeHudSettings, pickDisplay, snapToEdge } = require('./hud-geometry.cjs');

const hudEntry = path.join(__dirname, '..', 'renderer', 'hud.html');
const DRAG_LIMIT_MS = 60_000;
const SEND_CHANNELS = ['hud:pointer', 'hud:drag', 'hud:card'];
const INVOKE_CHANNELS = ['hud:initial', 'hud:refresh', 'hud:open-panel'];

function loadSettings(file) {
  try { return normalizeHudSettings(JSON.parse(fs.readFileSync(file, 'utf8'))); }
  catch { return normalizeHudSettings(null); }
}

function saveSettings(file, settings) {
  try {
    fs.mkdirSync(path.dirname(file), { recursive: true });
    const temporary = `${file}.${process.pid}.${randomUUID()}.tmp`;
    fs.writeFileSync(temporary, JSON.stringify(settings), 'utf8');
    fs.renameSync(temporary, file);
  } catch { /* a lost position is not worth interrupting the HUD */ }
}

function createHud({ settingsPath, onRefresh, onOpenPanel }) {
  let settings = loadSettings(settingsPath);
  let win = null;
  let layout = null;
  let items = [];
  let tokens = null;
  let drag = null;
  let ignoring = null;
  let recoveries = [];

  const alive = () => win && !win.isDestroyed();
  const send = (channel, payload) => { try { if (alive()) win.webContents.send(channel, payload); } catch { /* renderer restarting */ } };
  const layoutMessage = (dragging = false) => (layout ? { edge: layout.edge, strip: layout.strip, size: { width: layout.window.width, height: layout.window.height }, dragging } : null);
  const trusted = event => alive() && event.sender === win.webContents && event.senderFrame === win.webContents.mainFrame && isTrustedRenderer(event.senderFrame.url, hudEntry);

  function setIgnore(flag) {
    if (!alive() || ignoring === flag) return;
    ignoring = flag;
    if (flag) win.setIgnoreMouseEvents(true, { forward: true });
    else win.setIgnoreMouseEvents(false);
  }

  function place() {
    if (!alive() || drag) return;
    const display = pickDisplay(screen.getAllDisplays(), screen.getPrimaryDisplay(), settings.displayId);
    layout = layoutHud({ workArea: display.workArea, edge: settings.edge, offset: settings.offset });
    win.setBounds(layout.window, false);
    send('hud:layout', layoutMessage());
  }

  function startDrag() {
    if (drag || !alive() || !layout) return;
    const cursor = screen.getCursorScreenPoint();
    const [x, y] = win.getPosition();
    const startedAt = Date.now();
    drag = { dx: cursor.x - x, dy: cursor.y - y, timer: setInterval(() => {
      if (!alive() || Date.now() - startedAt > DRAG_LIMIT_MS) { endDrag(); return; }
      const point = screen.getCursorScreenPoint();
      win.setPosition(point.x - drag.dx, point.y - drag.dy, false);
    }, 16) };
    setIgnore(false);
    send('hud:layout', layoutMessage(true));
  }

  function endDrag() {
    if (!drag) return;
    clearInterval(drag.timer);
    drag = null;
    if (!alive() || !layout) return;
    const [x, y] = win.getPosition();
    const vertical = layout.edge === 'left' || layout.edge === 'right';
    const center = {
      x: x + layout.strip.x + (vertical ? STRIP.short : STRIP.long) / 2,
      y: y + layout.strip.y + (vertical ? STRIP.long : STRIP.short) / 2,
    };
    const display = screen.getDisplayNearestPoint({ x: Math.round(center.x), y: Math.round(center.y) });
    settings = { ...settings, displayId: display.id, ...snapToEdge({ workArea: display.workArea, point: center }) };
    saveSettings(settingsPath, settings);
    place();
  }

  function create() {
    win = new BrowserWindow({
      width: WINDOW_SIZE, height: WINDOW_SIZE, show: false, frame: false, transparent: true,
      backgroundColor: '#00000000', resizable: false, minimizable: false, maximizable: false,
      fullscreenable: false, skipTaskbar: true, focusable: false, hasShadow: false, alwaysOnTop: true,
      webPreferences: { preload: path.join(__dirname, 'preload-hud.cjs'), contextIsolation: true, nodeIntegration: false, sandbox: true },
    });
    const created = win;
    ignoring = null;
    win.setAlwaysOnTop(true, 'floating');
    setIgnore(true);
    win.webContents.setWindowOpenHandler(() => ({ action: 'deny' }));
    win.webContents.on('will-navigate', (event, url) => { if (!isTrustedRenderer(url, hudEntry)) event.preventDefault(); });
    win.webContents.on('will-attach-webview', event => event.preventDefault());
    win.webContents.session.setPermissionRequestHandler((_contents, _permission, callback) => callback(false));
    win.webContents.session.setPermissionCheckHandler(() => false);
    win.webContents.on('did-finish-load', () => {
      if (win !== created || !alive()) return;
      place();
      send('hud:state', items);
      if (tokens) send('hud:tokens', tokens);
      if (settings.visible) win.showInactive();
    });
    win.webContents.on('render-process-gone', (_event, details) => {
      if (details.reason === 'clean-exit') return;
      const now = Date.now();
      recoveries = recoveries.filter(time => now - time < 60000);
      if (recoveries.length >= 3) return;
      recoveries.push(now);
      setTimeout(() => { if (win === created && alive()) created.loadFile(hudEntry).catch(() => {}); }, 250);
    });
    win.on('closed', () => { if (win === created) { win = null; if (drag) { clearInterval(drag.timer); drag = null; } } });
    win.loadFile(hudEntry).catch(() => {});
  }

  ipcMain.on('hud:pointer', (event, inside) => { if (trusted(event) && typeof inside === 'boolean' && !drag) setIgnore(!inside); });
  ipcMain.on('hud:drag', (event, phase) => { if (!trusted(event)) return; if (phase === 'start') startDrag(); else if (phase === 'end') endDrag(); });
  ipcMain.on('hud:card', (event, open) => {
    if (!trusted(event) || typeof open !== 'boolean') return;
    win.setFocusable(open);
    if (open) win.focus();
  });
  const guard = handler => (event, ...args) => { if (!trusted(event)) throw new Error('不可信的请求来源'); return handler(...args); };
  ipcMain.handle('hud:initial', guard(() => ({ items, tokens, layout: layoutMessage() })));
  ipcMain.handle('hud:refresh', guard(async () => { await onRefresh(); return true; }));
  ipcMain.handle('hud:open-panel', guard(() => { onOpenPanel(); return true; }));

  const onDisplays = () => place();
  screen.on('display-metrics-changed', onDisplays);
  screen.on('display-added', onDisplays);
  screen.on('display-removed', onDisplays);

  if (settings.visible) create();

  return {
    isVisible: () => settings.visible,
    show() {
      settings = { ...settings, visible: true };
      saveSettings(settingsPath, settings);
      if (!alive()) create(); else { place(); win.showInactive(); }
    },
    hide() {
      settings = { ...settings, visible: false };
      saveSettings(settingsPath, settings);
      if (alive()) win.hide();
    },
    pushState(next) { items = Array.isArray(next) ? next : []; send('hud:state', items); },
    pushTokens(next) { tokens = next; send('hud:tokens', tokens); },
    window: () => (alive() ? win : null),
    destroy() {
      if (drag) { clearInterval(drag.timer); drag = null; }
      screen.removeListener('display-metrics-changed', onDisplays);
      screen.removeListener('display-added', onDisplays);
      screen.removeListener('display-removed', onDisplays);
      for (const channel of SEND_CHANNELS) ipcMain.removeAllListeners(channel);
      for (const channel of INVOKE_CHANNELS) ipcMain.removeHandler(channel);
      if (alive()) win.destroy();
      win = null;
    },
  };
}

module.exports = { createHud };
```

```js
(function () {
  'use strict';

  const SVG = 'http://www.w3.org/2000/svg';
  const QUARTER = 15 * 60000;
  const STRIP = { long: 158, short: 34 }; // hud-geometry.cjs
  const R = 10.25;

  let items = [];
  let tokens = null;
  let layout = { edge: 'right', strip: { x: 0, y: 0 }, size: { width: 480, height: 480 }, dragging: false };
  let openId = null;
  let showRate = false;
  let press = null;
  let inside = false;
  let pillTimer = null;
  let pillHide = null;
  let stripHide = null;

  const strip = document.querySelector('#strip');
  const pill = document.querySelector('#pill');
  const card = document.querySelector('#card');
  const pad = n => String(n).padStart(2, '0');
  const hm = d => `${pad(d.getHours())}:${pad(d.getMinutes())}`;
  const fmt = n => Math.round(n).toLocaleString('en-US');

  function el(tag, attrs = {}, children = []) {
    const node = tag.startsWith('svg:') ? document.createElementNS(SVG, tag.slice(4)) : document.createElement(tag);
    for (const [key, value] of Object.entries(attrs)) {
      if (value === null || value === undefined || value === false) continue;
      if (key === 'text') node.textContent = value;
      else node.setAttribute(key, value === true ? '' : String(value));
    }
    for (const child of [].concat(children)) if (child) node.append(child);
    return node;
  }

  function ring(item) {
    const used = item.ring.usedPct;
    return el('svg:svg', { class: `ring ${item.ring.level}${item.ring.dashed ? ' dashed' : ''}`, viewBox: '0 0 22 22', 'aria-hidden': 'true' }, [
      el('svg:circle', { class: 'track', cx: 11, cy: 11, r: R }),
      used === null ? null : el('svg:circle', { class: 'value', cx: 11, cy: 11, r: R, pathLength: 100, 'stroke-dasharray': `${Math.max(used, 1.5)} 100`, transform: 'rotate(-90 11 11)' }),
    ]);
  }

  const logo = (item, className) => el('img', { class: className, src: `../../assets/logos/${item.logo}`, alt: '', draggable: 'false' });

  function renderStrip() {
    strip.replaceChildren(...items.map(item => el('button', {
      class: `item${item.ring.dashed ? ' stale' : ''}`, type: 'button', 'data-id': item.id,
      'aria-label': `${item.name} ${item.label}`, 'aria-pressed': String(openId === item.id),
    }, [el('span', { class: 'mark' }, [ring(item), logo(item, 'logo')]), el('span', { class: `num ${item.ring.level}`, text: item.label })])));
  }

  function applyLayout(next) {
    if (!next) return;
    layout = next;
    document.body.className = `edge-${layout.edge}${layout.dragging ? ' dragging' : ''}`;
    const s = strip.style;
    s.left = s.right = s.top = s.bottom = '';
    const { x, y } = layout.strip;
    const { width, height } = layout.size;
    if (layout.edge === 'left') { s.left = `${x}px`; s.top = `${y}px`; }
    if (layout.edge === 'right') { s.right = `${width - x - STRIP.short}px`; s.top = `${y}px`; }
    if (layout.edge === 'top') { s.top = `${y}px`; s.left = `${x + STRIP.long / 2}px`; }
    if (layout.edge === 'bottom') { s.bottom = `${height - y - STRIP.short}px`; s.left = `${x + STRIP.long / 2}px`; }
    if (!layout.dragging) { hidePill(); closeCard(); }
  }

  // Put a floating node beside an anchor, on the side of the strip that faces into the screen.
  function place(node, anchor, gap) {
    const a = anchor.getBoundingClientRect();
    const s = strip.getBoundingClientRect();
    const n = node.getBoundingClientRect();
    const { width, height } = layout.size;
    const clampX = v => Math.max(4, Math.min(width - n.width - 4, v));
    const clampY = v => Math.max(4, Math.min(height - n.height - 4, v));
    let x; let y;
    if (layout.edge === 'left') { x = s.right + gap; y = clampY(a.top + a.height / 2 - n.height / 2); }
    else if (layout.edge === 'right') { x = s.left - gap - n.width; y = clampY(a.top + a.height / 2 - n.height / 2); }
    else if (layout.edge === 'top') { y = s.bottom + gap; x = clampX(a.left + a.width / 2 - n.width / 2); }
    else { y = s.top - gap - n.height; x = clampX(a.left + a.width / 2 - n.width / 2); }
    node.style.left = `${Math.round(x)}px`;
    node.style.top = `${Math.round(y)}px`;
  }

  const button = id => strip.querySelector(`[data-id="${id}"]`);

  function showPill(id) {
    const item = items.find(entry => entry.id === id);
    if (!item || openId || layout.dragging) return;
    pill.replaceChildren(...item.pill.map(seg => el('span', { class: `seg ${seg.level}` }, [seg.name ? el('b', { text: seg.name }) : null, el('span', { text: seg.text })])));
    pill.hidden = false;
    place(pill, button(id), 6);
  }

  function hidePill() {
    clearTimeout(pillTimer);
    clearTimeout(pillHide);
    pill.hidden = true;
  }

  const refreshIcon = () => el('svg:svg', { viewBox: '0 0 16 16', 'aria-hidden': 'true' }, [
    el('svg:path', { d: 'M13.2 8a5.2 5.2 0 1 1-1.5-3.7' }), el('svg:path', { d: 'M13.2 2.6v2.9h-2.9' }),
  ]);

  function bar(row) {
    const track = el('span', { class: `bar${row.usedPct === null ? ' unknown' : ''}` });
    if (row.usedPct !== null) {
      const fill = el('i', { class: row.level });
      fill.style.width = `${Math.max(row.usedPct, 2)}%`;
      track.append(fill);
    }
    return track;
  }

  function tokenSection(id) {
    const series = tokens?.[id];
    if (series?.status === 'unsupported') return null;
    const box = el('section', { class: 'tokens' });
    if (!series) { box.append(el('p', { class: 'muted', text: 'Token 读取中…' })); return box; }
    if (series.status !== 'ok') { box.append(el('p', { class: 'muted', text: 'Token 读取失败' })); return box; }
    const max = Math.max(1, ...series.buckets);
    const W = 312; const H = 44; const step = W / series.buckets.length;
    const chart = el('svg:svg', { class: 'chart', viewBox: `0 0 ${W} ${H}`, preserveAspectRatio: 'none', role: 'img', 'aria-label': `过去 24 小时共 ${fmt(series.total)} tokens` },
      series.buckets.map((n, i) => {
        if (n <= 0) return null;
        const h = Math.max(1, (n / max) * H);
        return el('svg:rect', { x: (i * step + 0.4).toFixed(2), width: (step - 0.8).toFixed(2), y: (H - h).toFixed(2), height: h.toFixed(2), rx: 0.8 });
      }));
    const tip = el('p', { class: 'tip', text: series.partial ? '部分文件本轮未读完' : '悬停柱子查看时段' });
    chart.addEventListener('mousemove', event => {
      const r = chart.getBoundingClientRect();
      const i = Math.max(0, Math.min(series.buckets.length - 1, Math.floor(((event.clientX - r.left) / r.width) * series.buckets.length)));
      const start = new Date(series.startAt + i * QUARTER);
      const end = new Date(start.getTime() + QUARTER);
      tip.textContent = `${start.getMonth() + 1}/${start.getDate()} ${hm(start)}–${hm(end)} · ${fmt(series.buckets[i])} tokens`;
    });
    box.append(el('div', { class: 'tokens-head' }, [el('h3', { text: 'Tokens · 24 小时' }), el('span', { text: fmt(series.total) })]), chart, tip);
    return box;
  }

  function renderCard() {
    const item = items.find(entry => entry.id === openId);
    if (!item) { closeCard(); return; }
    const c = item.card;
    const parts = [
      el('header', { class: 'card-head' }, [
        logo(item, 'card-logo'),
        el('div', { class: 'who' }, [el('h2', { text: item.name }), el('p', { text: c.plan })]),
        el('button', { class: 'icon', type: 'button', 'data-act': 'refresh', 'aria-label': '刷新', title: '刷新' }, [refreshIcon()]),
      ]),
      c.note ? el('p', { class: `note${c.stale ? ' stale' : ''}`, text: c.note }) : null,
      c.balance ? el('p', { class: 'balance', text: c.balance }) : null,
      c.rows.length ? el('div', { class: 'rows' }, c.rows.map(row => el('button', { class: 'row', type: 'button', 'data-act': 'rate', title: '点击切换 已用 / 消耗速率' }, [
        el('span', { class: 'row-label', text: row.label }),
        bar(row),
        el('span', { class: `row-value ${showRate ? 'none' : row.level}`, text: showRate ? row.burn : row.usedPct === null ? '–' : `${row.usedPct}%` }),
        el('span', { class: 'row-reset', text: row.reset }),
      ]))) : null,
      c.tokens ? tokenSection(item.id) : null,
      el('footer', { class: 'card-foot' }, [el('span', { text: c.freshness }), el('button', { class: 'link', type: 'button', 'data-act': 'panel', text: '完整面板' })]),
    ];
    card.replaceChildren(...parts.filter(Boolean));
    place(card, button(item.id), 8);
  }

  function openCard(id) {
    hidePill();
    openId = id;
    showRate = false;
    card.hidden = false;
    strip.classList.add('open');
    renderCard();
    renderStrip();
    window.hud.card(true);
  }

  function closeCard() {
    if (!openId) return;
    openId = null;
    showRate = false;
    card.hidden = true;
    renderStrip();
    window.hud.card(false);
  }

  function setItems(next) {
    items = Array.isArray(next) ? next : [];
    renderStrip();
    if (openId) renderCard();
  }

  strip.addEventListener('mouseenter', () => { clearTimeout(stripHide); strip.classList.add('open'); });
  strip.addEventListener('mouseleave', () => {
    stripHide = setTimeout(() => { if (!openId) strip.classList.remove('open'); }, 250);
    clearTimeout(pillTimer);
    pillHide = setTimeout(hidePill, 200);
  });
  strip.addEventListener('mouseover', event => {
    const target = event.target.closest('.item');
    if (!target || press?.dragging || openId) return;
    clearTimeout(pillHide);
    clearTimeout(pillTimer);
    if (!pill.hidden) showPill(target.dataset.id);
    else pillTimer = setTimeout(() => showPill(target.dataset.id), 150);
  });

  // Press, then more than 4px of movement, is a drag; otherwise it is a click.
  strip.addEventListener('pointerdown', event => {
    if (event.button !== 0) return;
    press = { x: event.screenX, y: event.screenY, id: event.target.closest('.item')?.dataset.id || null, dragging: false };
    try { strip.setPointerCapture(event.pointerId); } catch { /* synthetic events have no capture */ }
  });
  strip.addEventListener('pointermove', event => {
    if (!press || press.dragging) return;
    if (Math.hypot(event.screenX - press.x, event.screenY - press.y) > 4) {
      press.dragging = true;
      hidePill();
      closeCard();
      window.hud.drag('start');
    }
  });
  strip.addEventListener('pointerup', event => {
    if (!press) return;
    const done = press;
    press = null;
    try { strip.releasePointerCapture(event.pointerId); } catch { /* not captured */ }
    if (done.dragging) window.hud.drag('end');
    else if (done.id) { if (openId === done.id) closeCard(); else openCard(done.id); }
  });
  strip.addEventListener('pointercancel', () => {
    if (press?.dragging) window.hud.drag('end');
    press = null;
  });

  card.addEventListener('click', event => {
    const action = event.target.closest('[data-act]');
    if (!action) return;
    if (action.dataset.act === 'rate') { showRate = !showRate; renderCard(); }
    if (action.dataset.act === 'panel') { window.hud.openPanel().catch(() => {}); closeCard(); }
    if (action.dataset.act === 'refresh') {
      action.disabled = true;
      window.hud.refresh().catch(() => {}).finally(() => { action.disabled = false; });
    }
  });

  document.addEventListener('keydown', event => { if (event.key === 'Escape') closeCard(); });
  window.addEventListener('blur', closeCard);

  // The window ignores the mouse except over visible parts; forwarded moves tell us when to stop ignoring.
  document.addEventListener('mousemove', event => {
    const now = Boolean(press) || Boolean(event.target.closest?.('.hit'));
    if (now !== inside) { inside = now; window.hud.pointer(now); }
  });
  document.documentElement.addEventListener('mouseleave', () => {
    if (inside && !press) { inside = false; window.hud.pointer(false); }
  });

  window.hud.onState(setItems);
  window.hud.onTokens(next => { tokens = next; if (openId) renderCard(); });
  window.hud.onLayout(applyLayout);
  window.hud.initial().then(state => {
    applyLayout(state.layout);
    tokens = state.tokens;
    setItems(state.items);
  }).catch(() => {});
})();
```
